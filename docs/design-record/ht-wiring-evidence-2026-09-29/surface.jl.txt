# HT's derived public surface as the harness contracts read it, the Phenomenology modules left out.
using GBGHomogeneousTurbulence, GBGCodingTools
const HT = GBGHomogeneousTurbulence; const TH = GBGCodingTools.TestHarness
function submodules(m, acc = Module[])
    for n in names(m; all = true)
        isdefined(m, n) || continue
        v = getfield(m, n)
        v isa Module && v !== m && parentmodule(v) === m && !(v in acc) && (push!(acc, v); submodules(v, acc))
    end
    return acc
end
all_mods = submodules(HT)
kept = [m for m in all_mods if !(:Phenomenology in fullname(m))]
println("modules: ", length(all_mods), " in all, ", length(kept), " without Phenomenology: ", join(nameof.(kept), ", "))
for (label, mods) in (("without Phenomenology", kept), ("all", all_mods))
    ops = TH.package_operations(HT, mods); tys = TH.struct_types(mods)
    inplace = count(n -> endswith(String(n), "!"), ops)
    println(label, ": ", length(ops), " operations, ", inplace, " in-place, ", length(ops) - inplace, " value-returning; ", length(tys), " types")
end
