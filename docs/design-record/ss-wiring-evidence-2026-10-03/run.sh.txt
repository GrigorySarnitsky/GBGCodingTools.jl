#!/bin/bash
# One rehearsal of SS's first wiring at v2.0.0-rc.57, every step logged; the steps SS-WIRING.md gives.
S=/tmp/claude-1000/-home-grigory-Dropbox-Research-and-Teach-AI-Helpers-GBGCodingTools/7683c89e-093e-4433-8205-eb60c644023a/scratchpad
R=$S/ss-wiring; L=$R/log; SS="/home/grigory/Dropbox/Research and Teach/Statistics/GBGStreamingStatistics"
V=2.0.0-rc.57
rm -rf $R/ss $R/ss13 $R/ss11 $L; mkdir -p $L
git clone -q "$SS" $R/ss && cd $R/ss || exit 9
git remote set-url origin https://github.com/GrigorySarnitsky/GBGStreamingStatistics.jl.git
git remote set-url --push origin DISABLED
# SS's own checkout has ignored docs and test manifests (neither holds the tool); the clone gets copies.
cp "$SS/docs/Manifest.toml" docs/Manifest.toml; cp "$SS/test/Manifest.toml" test/Manifest.toml
step() { local n=$1; shift; "$@" > $L/$n.out 2>&1; local e=$?; echo "$n exit $e" >> $L/steps.txt; }
echo "HEAD $(git log -1 --format=%h)" > $L/steps.txt
step 00-install gbg --dev install
step 01-plain-init gbg init
step 02-init gbg --dev init
git status --short > $L/02a-status.txt
step 03-bump gbg --dev bump $V
cp docs/Project.toml $R/docs-project-after-bump.toml; cp .gbg/config.jl $R/config-after-bump.jl
step 04-docs-project python3 $R/edits.py project
diff $R/docs-project-after-bump.toml docs/Project.toml > $L/04a-project-diff.txt
step 04b-resolve julia +1.12 --project=docs --startup-file=no -e 'using Pkg; Pkg.resolve()'
sed -n '/^\[\[deps.GBGCodingTools\]\]/,/^$/p;/^\[\[deps.GBGStreamingStatistics\]\]/,/^$/p' docs/Manifest.toml > $L/04c-manifest.txt
step 05-sync gbg sync
step 06-check-before-edits gbg check
python3 $R/edits.py entry && python3 $R/edits.py workflows; echo "edits exit $?" >> $L/steps.txt
git diff -- docs/make.jl test/runtests.jl > $R/entry.diff; git diff -- .github/workflows > $R/workflows.diff
step 07-check-after-edits gbg check
cp $R/config-final.jl .gbg/config.jl; diff $R/config-after-bump.jl .gbg/config.jl > $L/08a-config-diff.txt
step 08-check-after-config gbg check
step 09-sync gbg sync
step 10-check-after-sync gbg check
step 11-rebaseline gbg check --rebaseline
wc -l docs/prose_shape.csv > $L/11a-baseline-rows.txt
step 12-sync-check gbg sync --check
step 13-test julia +1.12 --project --startup-file=no -e 'using Pkg; Pkg.test()'
step 14-docs julia +1.12 --project=docs --startup-file=no docs/make.jl
sed 's/^GBGCodingTools.makedocs(ROOT;/Documenter.makedocs(;/; s/^deploydocs(/false \&\& deploydocs(/' docs/make.jl > docs/make-plain.jl
step 15-plain-build julia +1.12 --project=docs --startup-file=no docs/make-plain.jl
rm docs/make-plain.jl
step 16-site julia +1.12 --project=docs --startup-file=no -e 'using Documenter, GBGCodingTools; E = Base.get_extension(GBGCodingTools, :GBGCodingToolsDocumenterExt); exit(E.rules_and_wiring(pwd(); site = joinpath(pwd(), "docs", "build")))'
git status --short > $L/17-status-before-commit.txt; git diff --stat > $L/17-diffstat.txt
git ls-files .claude | wc -l > $L/17-claude-count.txt
git add -u && git add .claude/hooks/ .claude/settings.json .claude/skills/ .gbg/ .github/actions/ docs/prose_shape.csv test/regressions/README.md
git -c user.name=rehearsal -c user.email=r@example.invalid commit -q -m "Wire GBGCodingTools v$V" -m "Co-Authored-By: Anthropic Claude <noreply@anthropic.com>"; echo "18-commit exit $?" >> $L/steps.txt
git status --short > $L/18-status-after-commit.txt
printf '{"tool_input":{"command":"git push"},"cwd":"%s"}' "$PWD" | CLAUDE_PROJECT_DIR="$PWD" bash .claude/hooks/push-gate.sh > $L/19-hook.out 2>&1; echo "19-hook exit $?" >> $L/steps.txt
step 20-check-committed gbg check
step 20a-surface julia +1.12 --project=docs --startup-file=no $R/surface.jl
step 21-sync-check-committed gbg sync --check
step 21a-leak gbg leak
# SS's CI lane '1': the current stable, four threads, no manifests, the docs environment by the workflow's step.
cp -r $R/ss $R/ss13 && cd $R/ss13 && rm -f docs/Manifest.toml test/Manifest.toml
step 22-docs-env-113 julia +1.13 --project=docs --startup-file=no -e 'using Pkg; Pkg.instantiate()'
step 23-docs-113 julia +1.13 --project=docs --startup-file=no docs/make.jl
step 24-test-113 env GIT_CONFIG_GLOBAL=/dev/null JULIA_NUM_THREADS=4 julia +1.13 --project --startup-file=no -e 'using Pkg; Pkg.test()'
git status --short > $L/25-status-113.txt
# SS's CI lane '1.11', its compat floor: the tool needs 1.12.
cp -r $R/ss $R/ss11 && cd $R/ss11 && rm -f docs/Manifest.toml test/Manifest.toml
step 26-test-111 julia +1.11 --project --startup-file=no -e 'using Pkg; Pkg.test()'
step 27-docs-env-111 julia +1.11 --project=docs --startup-file=no -e 'using Pkg; Pkg.instantiate()'
cd $R/ss
echo done >> $L/steps.txt
