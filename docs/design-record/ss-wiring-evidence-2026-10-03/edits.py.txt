# The edits SS-WIRING.md gives for Step 2 (docs/Project.toml) and Step 3 (make.jl, runtests.jl, workflows).
import sys
which = sys.argv[1]
def sub(p, a, b, n=1):
    t = open(p).read(); assert a in t, (p, a[:50]); open(p, "w").write(t.replace(a, b, n))
if which == "project":
    p = "docs/Project.toml"; t = open(p).read()
    assert "[sources]" in t
    sub(p, "[sources]\n", "[sources]\nGBGStreamingStatistics = {path = \"..\"}\n")
if which == "entry":
    p = "docs/make.jl"
    sub(p, "using Documenter\nusing GBGStreamingStatistics\n",
        "using Documenter\nusing GBGCodingTools\nusing GBGStreamingStatistics\nconst ROOT = normpath(joinpath(@__DIR__, \"..\"))\nconst TOOLING = GBGCodingTools.Render.pages(GBGCodingTools.ConfigIO.read_config(ROOT), ROOT)\n")
    sub(p, "makedocs(\n  sitename  = \"GBGStreamingStatistics.jl\",", "GBGCodingTools.makedocs(ROOT;\n  sitename  = \"GBGStreamingStatistics.jl\",")
    sub(p, "      \"Internals\"     => \"internals.md\",\n", "      \"Internals\"     => \"internals.md\",\n      \"Documentation tooling\" => only(TOOLING),\n")
    p = "test/runtests.jl"
    open(p, "a").write("\nusing GBGCodingTools\nGBGCodingTools.Engine.Harness.test(dirname(@__DIR__); checks = (GBGCodingTools.Render.wiring,))\n")
if which == "workflows":
    auth = "      - uses: actions/checkout@v6\n      - uses: ./.github/actions/gbg-auth\n        with:\n          token: ${{ secrets.GBG_DEPS_TOKEN }}\n"
    sub(".github/workflows/CI.yml", "      - uses: actions/checkout@v6\n", auth)
    p = ".github/workflows/docs.yml"
    sub(p, "      - uses: actions/checkout@v6\n", auth)
    sub(p, "          julia --color=yes --project=docs -e '\n            using Pkg;\n            Pkg.develop(PackageSpec(path=pwd()));\n            Pkg.instantiate();\n          '\n",
        "          julia --color=yes --project=docs -e 'using Pkg; Pkg.instantiate()'\n")
