#!/bin/bash
# One rehearsal of HT's first wiring at v2.0.0-rc.35, every step logged; the steps HT-WIRING.md gives.
S=/tmp/claude-1000/-home-grigory-Dropbox-Research-and-Teach-AI-Helpers-GBGCodingTools/7683c89e-093e-4433-8205-eb60c644023a/scratchpad
R=$S/r6; L=$R/log; HT="/home/grigory/Dropbox/Research and Teach/Isotropic Turbulence/GBGHomogeneousTurbulence"
rm -rf $R/ht $L; mkdir -p $L
git clone -q "$HT" $R/ht && cd $R/ht || exit 9
git remote set-url origin https://github.com/GrigorySarnitsky/GBGHomogeneousTurbulence.jl.git
git remote set-url --push origin DISABLED
# HT's own checkout has an ignored docs manifest (which does not hold the tool); the clone gets a copy.
cp "$HT/docs/Manifest.toml" docs/Manifest.toml
step() { local n=$1; shift; "$@" > $L/$n.out 2>&1; local e=$?; echo "$n exit $e" >> $L/steps.txt; }
echo "HEAD $(git log -1 --format=%h)" > $L/steps.txt
step 01-plain-init gbg init
step 02-init gbg --dev init
step 03-bump gbg --dev bump 2.0.0-rc.35
cp docs/Project.toml $R/docs-project-after-bump.toml
step 04-docs-env julia --project=docs --startup-file=no -e 'using Pkg; Pkg.add(url="https://github.com/GrigorySarnitsky/GBGPeriodicFields.jl"); Pkg.develop(PackageSpec(path=pwd())); Pkg.instantiate()'
sed -n '/^\[\[deps.GBGCodingTools\]\]/,/^$/p' docs/Manifest.toml > $L/04b-manifest.txt
diff $R/docs-project-after-bump.toml docs/Project.toml > $L/04c-project-diff.txt
step 05-sync gbg sync
step 06-check-before-edits gbg check
git apply $R/entry.diff && git apply $R/workflows.diff; echo "edits exit $?" >> $L/steps.txt
step 07-check-after-edits gbg check
cp $R/config-final.jl .gbg/config.jl
step 08-check-after-config gbg check
step 09-sync gbg sync
step 10-check-after-sync gbg check
step 11-rebaseline gbg check --rebaseline
step 12-sync-check gbg sync --check
step 13-test julia --project --startup-file=no -e 'using Pkg; Pkg.test()'
step 14-docs julia --project=docs --startup-file=no docs/make.jl
sed 's/^GBGCodingTools.makedocs(ROOT;/Documenter.makedocs(;/; s/^deploydocs(/false \&\& deploydocs(/' docs/make.jl > docs/make-plain.jl
step 15-plain-build julia --project=docs --startup-file=no docs/make-plain.jl
rm docs/make-plain.jl
step 16-site julia --project=docs --startup-file=no -e 'using Documenter, GBGCodingTools; E = Base.get_extension(GBGCodingTools, :GBGCodingToolsDocumenterExt); exit(E.rules_and_wiring(pwd(); site = joinpath(pwd(), "docs", "build")))'
# What the wiring commit holds: the docs project as bump wrote it, the environment step's two lines dropped.
cp $R/docs-project-after-bump.toml docs/Project.toml
step 17-check-committed-project gbg check
step 18-sync-check-committed gbg sync --check
git status --short > $L/19-status.txt; git diff --stat > $L/19-diffstat.txt
git ls-files .claude | wc -l > $L/20-claude-count.txt
echo done >> $L/steps.txt
